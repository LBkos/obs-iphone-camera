cmake_minimum_required(VERSION 3.16)
project(obs-iphone-camera VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(ENABLE_BONJOUR "Advertise the receiver via mDNS (dns_sd / Avahi compat)" ON)
option(BUILD_RECEIVER_TEST "Build the standalone receiver test tool (no OBS needed)" ON)
option(BUILD_PLUGIN "Build the OBS plugin" ON)

find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil)

# ---------------------------------------------------------------- mDNS
set(DNSSD_TARGET "")
if(ENABLE_BONJOUR)
  if(APPLE)
    # dns_sd is part of libSystem.
    set(HAVE_DNSSD ON)
  else()
    pkg_check_modules(DNSSD IMPORTED_TARGET avahi-compat-libdns_sd)
    if(DNSSD_FOUND)
      set(HAVE_DNSSD ON)
      set(DNSSD_TARGET PkgConfig::DNSSD)
    else()
      message(WARNING "avahi-compat-libdns_sd not found: building without Bonjour. "
                      "Install libavahi-compat-libdnssd-dev (Debian/Ubuntu) or avahi-compat-libdns_sd-devel (Fedora), "
                      "or enter the IP address in the iOS app manually.")
    endif()
  endif()
endif()

# Network/protocol/decoder core, independent from libobs.
add_library(obscam-core STATIC src/receiver.c src/discovery.c src/h264-decoder.c)
set_target_properties(obscam-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(obscam-core PUBLIC src)
target_link_libraries(obscam-core PUBLIC PkgConfig::FFMPEG Threads::Threads ${DNSSD_TARGET})
if(HAVE_DNSSD)
  target_compile_definitions(obscam-core PRIVATE HAVE_DNSSD)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(obscam-core PRIVATE -Wall -Wextra)
endif()

if(BUILD_RECEIVER_TEST)
  add_executable(obscam-receiver-test tools/receiver-test.c)
  target_link_libraries(obscam-receiver-test PRIVATE obscam-core)
endif()

if(NOT BUILD_PLUGIN)
  return()
endif()

# ---------------------------------------------------------------- libobs
# 1) Distro / OBS SDK package (libobs-dev, obs-studio-devel) -> find_package
# 2) Fallback: libobs headers from an obs-studio source checkout (OBS_SOURCE_DIR)
#    and link against the installed libobs at runtime.
find_package(libobs QUIET)
add_library(obs-iphone-camera MODULE src/plugin-main.c)
set_target_properties(obs-iphone-camera PROPERTIES PREFIX "")
target_link_libraries(obs-iphone-camera PRIVATE obscam-core)

if(TARGET OBS::libobs)
  message(STATUS "Using installed libobs package")
  target_link_libraries(obs-iphone-camera PRIVATE OBS::libobs)
else()
  set(OBS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/obs-studio" CACHE PATH "obs-studio source checkout")
  if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h")
    message(FATAL_ERROR "libobs not found. Install the OBS development package, or run ./scripts/fetch-obs-headers.sh, "
                        "or pass -DOBS_SOURCE_DIR=/path/to/obs-studio")
  endif()
  message(STATUS "Using libobs headers from ${OBS_SOURCE_DIR}")
  # obsconfig.h is normally generated by OBS's own build.
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obs-generated/obsconfig.h"
       "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  target_include_directories(obs-iphone-camera PRIVATE "${OBS_SOURCE_DIR}/libobs"
                                                       "${CMAKE_CURRENT_BINARY_DIR}/obs-generated")
  # On non-x86 CPUs (Apple Silicon, ARM Linux) libobs headers need SIMDe.
  find_path(SIMDE_INCLUDE_DIR simde/x86/sse2.h PATHS /opt/homebrew/include /usr/local/include /usr/include)
  if(SIMDE_INCLUDE_DIR)
    target_include_directories(obs-iphone-camera PRIVATE "${SIMDE_INCLUDE_DIR}")
  elseif(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i.86")
    message(FATAL_ERROR "SIMDe headers are required on ${CMAKE_SYSTEM_PROCESSOR}: install libsimde-dev / simde")
  endif()
  find_library(LIBOBS_LIBRARY NAMES obs libobs PATHS /usr/lib /usr/local/lib /usr/lib64
                                                     /usr/lib/x86_64-linux-gnu /usr/lib/aarch64-linux-gnu)
  if(LIBOBS_LIBRARY)
    target_link_libraries(obs-iphone-camera PRIVATE "${LIBOBS_LIBRARY}")
  elseif(APPLE)
    # Symbols are resolved from libobs.framework when OBS loads the plugin.
    target_link_options(obs-iphone-camera PRIVATE -undefined dynamic_lookup)
  else()
    message(STATUS "libobs.so not found; leaving libobs symbols to be resolved at load time")
  endif()
endif()

# ---------------------------------------------------------------- install
if(APPLE)
  # OBS 28+ on macOS loads plugin bundles: <name>.plugin/Contents/{MacOS,Resources}
  set_target_properties(obs-iphone-camera PROPERTIES
    BUNDLE ON
    BUNDLE_EXTENSION plugin
    MACOSX_BUNDLE_GUI_IDENTIFIER com.obscamera.obs-iphone-camera
    MACOSX_BUNDLE_BUNDLE_NAME obs-iphone-camera
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
  add_custom_command(TARGET obs-iphone-camera POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data"
            "$<TARGET_BUNDLE_CONTENT_DIR:obs-iphone-camera>/Resources")
  install(TARGETS obs-iphone-camera LIBRARY DESTINATION . BUNDLE DESTINATION .)
else()
  # Per-user layout: ~/.config/obs-studio/plugins/obs-iphone-camera/{bin/64bit,data}
  # Use -DCMAKE_INSTALL_PREFIX=~/.config/obs-studio/plugins/obs-iphone-camera
  install(TARGETS obs-iphone-camera LIBRARY DESTINATION bin/64bit)
  install(DIRECTORY data/ DESTINATION data)
endif()
